cmake_minimum_required(VERSION 3.22)
project(ukmla_reader)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)

if(ANDROID_ABI STREQUAL "arm64-v8a")
    set(GGML_VULKAN ON CACHE BOOL "" FORCE)
    if(NOT DEFINED Vulkan_INCLUDE_DIR OR NOT EXISTS "${Vulkan_INCLUDE_DIR}/vulkan/vulkan.hpp")
        message(FATAL_ERROR "ARM64 Vulkan build requires pinned Vulkan-Hpp at Vulkan_INCLUDE_DIR")
    endif()
    if(NOT DEFINED Vulkan_LIBRARY OR NOT EXISTS "${Vulkan_LIBRARY}")
        message(FATAL_ERROR "ARM64 Vulkan build requires the Android NDK Vulkan loader")
    endif()
    if(NOT DEFINED Vulkan_GLSLC_EXECUTABLE OR NOT EXISTS "${Vulkan_GLSLC_EXECUTABLE}")
        message(FATAL_ERROR "ARM64 Vulkan build requires host glslc")
    endif()
    message(STATUS "UKMLA Vulkan headers: ${Vulkan_INCLUDE_DIR}")
    message(STATUS "UKMLA Vulkan loader: ${Vulkan_LIBRARY}")
    message(STATUS "UKMLA glslc: ${Vulkan_GLSLC_EXECUTABLE}")
else()
    set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
    message(STATUS "UKMLA Vulkan disabled for ABI ${ANDROID_ABI}; emulator smoke tests use CPU")
endif()

set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
add_subdirectory(../../../../vendor/llama.cpp llama)

if(ANDROID_ABI STREQUAL "arm64-v8a")
    if(NOT TARGET Vulkan::Vulkan)
        message(FATAL_ERROR "CMake did not create Vulkan::Vulkan for the ARM64 reader build")
    endif()
    get_target_property(UKMLA_VULKAN_TARGET_INCLUDES Vulkan::Vulkan INTERFACE_INCLUDE_DIRECTORIES)
    message(STATUS "Vulkan::Vulkan include dirs: ${UKMLA_VULKAN_TARGET_INCLUDES}")
    if(NOT TARGET ggml-vulkan)
        message(FATAL_ERROR "llama.cpp did not create the ggml-vulkan backend")
    endif()
endif()

add_library(reader SHARED reader.cpp)
target_link_libraries(reader llama android log)
