cmake_minimum_required(VERSION 3.22)
project(ukmla_reader)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)

option(UKMLA_READER_EXPERIMENTAL_VULKAN "Enable experimental Vulkan acceleration" OFF)
if(UKMLA_READER_EXPERIMENTAL_VULKAN AND ANDROID_ABI STREQUAL "arm64-v8a")
    set(GGML_VULKAN ON CACHE BOOL "" FORCE)
    if(NOT DEFINED Vulkan_INCLUDE_DIR OR NOT EXISTS "${Vulkan_INCLUDE_DIR}/vulkan/vulkan.hpp")
        message(FATAL_ERROR "Experimental Vulkan requires pinned Vulkan-Hpp at Vulkan_INCLUDE_DIR")
    endif()
    if(NOT DEFINED SPIRV_INCLUDE_DIR OR NOT EXISTS "${SPIRV_INCLUDE_DIR}/spirv/unified1/spirv.hpp")
        message(FATAL_ERROR "Experimental Vulkan requires pinned SPIR-V headers at SPIRV_INCLUDE_DIR")
    endif()
    if(NOT DEFINED Vulkan_LIBRARY OR NOT EXISTS "${Vulkan_LIBRARY}")
        message(FATAL_ERROR "Experimental Vulkan requires the Android NDK Vulkan loader")
    endif()
    if(NOT DEFINED Vulkan_GLSLC_EXECUTABLE OR NOT EXISTS "${Vulkan_GLSLC_EXECUTABLE}")
        message(FATAL_ERROR "Experimental Vulkan requires host glslc")
    endif()
    message(STATUS "UKMLA experimental Vulkan enabled")
else()
    set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
    message(STATUS "UKMLA stable reader build: Vulkan disabled; CPU inference only")
endif()

set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
add_subdirectory(../../../../vendor/llama.cpp llama)

if(UKMLA_READER_EXPERIMENTAL_VULKAN AND ANDROID_ABI STREQUAL "arm64-v8a")
    if(NOT TARGET ggml-vulkan)
        message(FATAL_ERROR "llama.cpp did not create the experimental ggml-vulkan backend")
    endif()
    target_include_directories(ggml-vulkan SYSTEM PRIVATE "${SPIRV_INCLUDE_DIR}")
endif()

add_library(reader SHARED reader.cpp)
target_link_libraries(reader llama android log)
