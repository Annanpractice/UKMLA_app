cmake_minimum_required(VERSION 3.22)
project(gpu_probe)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(flag GGML_NATIVE GGML_OPENMP GGML_LLAMAFILE LLAMA_BUILD_TESTS LLAMA_BUILD_EXAMPLES LLAMA_BUILD_TOOLS LLAMA_BUILD_SERVER LLAMA_CURL LLAMA_OPENSSL)
 set(${flag} OFF CACHE BOOL "" FORCE)
endforeach()
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_OPENCL OFF CACHE BOOL "" FORCE)
if(PROBE_BACKEND STREQUAL "vulkan")
 set(GGML_VULKAN ON CACHE BOOL "" FORCE)
elseif(PROBE_BACKEND STREQUAL "opencl")
 set(GGML_OPENCL ON CACHE BOOL "" FORCE)
 set(GGML_OPENCL_USE_ADRENO_KERNELS ON CACHE BOOL "" FORCE)
elseif(PROBE_BACKEND STREQUAL "opencl_generic")
 set(GGML_OPENCL ON CACHE BOOL "" FORCE)
 set(GGML_OPENCL_USE_ADRENO_KERNELS OFF CACHE BOOL "" FORCE)
endif()
add_subdirectory(${LLAMA_SOURCE} llama)
if(TARGET ggml-vulkan)
 target_include_directories(ggml-vulkan SYSTEM PRIVATE ${SPIRV_INCLUDE_DIR})
endif()
add_library(probe_${PROBE_BACKEND} SHARED probe.cpp)
target_compile_definitions(probe_${PROBE_BACKEND} PRIVATE PROBE_BACKEND="${PROBE_BACKEND}")
target_link_libraries(probe_${PROBE_BACKEND} llama android log)

if(PROBE_BACKEND STREQUAL "vulkan")
 target_compile_definitions(probe_vulkan PRIVATE PROBE_VULKAN)
 target_include_directories(probe_vulkan PRIVATE ${Vulkan_INCLUDE_DIR})
 target_link_libraries(probe_vulkan ${Vulkan_LIBRARY})
endif()
